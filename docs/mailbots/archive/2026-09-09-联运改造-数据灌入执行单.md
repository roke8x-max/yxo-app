## 1. 背景（实查证据）

```
route_draft / route_waybill / route_dsk / route_atb / route_tracing   (routing.py)
```

## 3. 格式铁律（写错会静默不转发）

- `to_addrs` / `cc_addrs` 是 **JSON 数组文本**，例如 `'["gongqilin@cqjzxly.cn", ...]'`。
- 两个库都用 `json.loads` 解析（`store.py:121` / 旧系统 `Dsk_Robot.py:416`）。写成逗号串会 `json.loads` 失败 → 解析成空 → 静默 `no_route`。
- 用 `INSERT OR REPLACE` + `UNIQUE(bot, scope, key)` 保证幂等（重跑不会重复行）。

## 4. 执行脚本（新系统，必做 · 上线硬前置）

> 路径默认写死生产值；若部署目录不同，改 `NEXT_BOT_CONFIG_DB` / `YXO_DB`。
> **默认 dry-run（只打印要写什么，不落库）；加 `--apply` 才真正写入。**
> 需先确保能访问 `\\10.0.199.184\yxo_data\yxo_app\data\yxo.db`。

```python
# scripts/seed_next_bot_config.py
import sys, os, json, sqlite3, argparse
REPO = r"D:\YXO_DATA\yxo_app"                      # 生产 yxo_app 仓库根
sys.path.insert(0, REPO)
import mailbots_next.core.store as store           # import 即触发 init_bot_config_db() 建正确表

YXO_DB      = r"\\10.0.199.184\yxo_data\yxo_app\data\yxo.db"
NEXT_BOT_CONFIG_DB = os.path.join(REPO, "mailbots_next", "data", "bot_config.db")

# 联运：洋确认的真实收件人（非占位）
LIANYUN_TO = ["gongqilin@cqjzxly.cn", "yuyanling@cqjzxly.cn", "jjb@cqjzxly.cn", "wanglu@cqjzxly.cn"]
LIANYUN_CC = ["3841559246@qq.com"]

def mirror_existing(src_db):
    """从生产 yxo.db 的 bot_config(scope='company') 镜像现有公司收件人，按 key 合并各 bot 的 to/cc。"""
    con = sqlite3.connect(f"file:{src_db}?mode=ro", uri=True)
    cur = con.cursor()
    cur.execute("SELECT bot, scope, key, to_addrs, cc_addrs FROM bot_config WHERE scope='company'")
    agg = {}
    for bot, scope, key, to_s, cc_s in cur.fetchall():
        if key == "联运":
            continue                                  # 联运用下方显式值，不走镜像
        a = agg.setdefault(key, {"to": set(), "cc": set()})
        try:
            a["to"] |= set(json.loads(to_s or "[]"))
            a["cc"] |= set(json.loads(cc_s or "[]"))
        except Exception as e:
            print(f"  [WARN] 跳过 {key}({bot}) 解析失败: {e}")
    con.close()
    return agg

def main():
    ap = argparse.ArgumentParser()
    ap.add_argument("--apply", action="store_true", help="真正写入；不加只打印")
    args = ap.parse_args()

    store.init_bot_config_db()                        # 确保新系统表结构正确（含 UNIQUE）
    companies = mirror_existing(YXO_DB)
    companies["联运"] = {"to": set(LIANYUN_TO), "cc": set(LIANYUN_CC)}

    print(f"将写入 {len(companies)} 家 company 行到: {NEXT_BOT_CONFIG_DB}")
    for k, v in companies.items():
        print(f"  {k}: to={sorted(v['to'])} cc={sorted(v['cc'])}")

    if not args.apply:
        print("\n[DRY-RUN] 未写入。确认无误后加 --apply 执行。")
        return

    conn = sqlite3.connect(NEXT_BOT_CONFIG_DB, timeout=30)
    n = 0
    for key, v in companies.items():
        conn.execute(
            """INSERT OR REPLACE INTO bot_config (bot, scope, key, to_addrs, cc_addrs, extra, updated_at)
               VALUES ('all', 'company', ?, ?, ?, '{}', datetime('now'))""",
            (key, json.dumps(sorted(v["to"])), json.dumps(sorted(v["cc"]))),
        )
        n += 1
    conn.commit(); conn.close()
    print(f"\n[OK] seeded {n} company rows")

if __name__ == "__main__":
    main()
```

**预期输出（加 `--apply` 后）**：`seeded 8 company rows`（7 家镜像 + 联运）。
